<script>
    // Load saved data from localStorage or set defaults
    let appData = JSON.parse(localStorage.getItem('subMeterData')) || {
        baseReading: 0,
        rechargeUnits: 0,
        history: []
    };

    function updateDisplay() {
        document.getElementById('dispRecharge').innerText = appData.rechargeUnits.toFixed(2) + ' kWh';
        document.getElementById('dispBaseReading').innerText = appData.baseReading.toFixed(2) + ' kWh';

        // Calculate consumed if we have a base reading recorded
        let consumed = 0;
        let balance = appData.rechargeUnits;

        if(appData.history.length > 0) {
            // Estimate consumption based on latest record if needed, 
            // Or prompt user for live reading check. Let's make it calculate live if input exists.
        }
        
        // Render history items (up to 2 rest cumulations)
        let historyHTML = '';
        if (appData.history.length === 0) {
            historyHTML = '<p style="font-size: 0.85rem; color: #64748b; text-align: center;">No history yet.</p>';
        } else {
            appData.history.slice(0, 2).forEach((item, index) => {
                historyHTML += `
                    <div class="history-item">
                        <strong>#${index + 1} Recharge:</strong> +${item.units} kWh 
                        <br><span style="color: #64748b;">Meter was at: ${item.meterReading} kWh (${item.date})</span>
                    </div>
                `;
            });
        }
        document.getElementById('historyList').innerHTML = historyHTML;
    }

    document.getElementById('rechargeForm').addEventListener('submit', function(e) {
        e.preventDefault();
        
        const currentMeter = parseFloat(document.getElementById('currentMeter').value);
        const rechargeUnits = parseFloat(document.getElementById('rechargeUnits').value);

        // Save current state into history
        const newRecord = {
            meterReading: currentMeter,
            units: rechargeUnits,
            date: new Date().toLocaleDateString()
        };

        appData.history.unshift(newRecord); // Add to the beginning
        appData.baseReading = currentMeter;
        appData.rechargeUnits = rechargeUnits;

        // Save to browser memory
        localStorage.setItem('subMeterData', JSON.stringify(appData));

        // Reset form & update view
        document.getElementById('rechargeForm').reset();
        updateDisplay();
        alert('Recharge logged successfully!');
    });

    // Initial load
    updateDisplay();
</script>
